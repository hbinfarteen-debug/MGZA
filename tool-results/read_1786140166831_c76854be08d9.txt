   324→   324→// HOVER TIP COMPONENT
   325→   325→// Uses position:fixed + getBoundingClientRect to escape overflow:auto clipping.
   326→   326→// Always renders wrapper div to prevent layout shift when toggling.
   327→   327→// ═══════════════════════════════════════════════════════
   328→   328→
   329→   329→function HoverTip({ tipId, children, screenId }: { tipId: string; children: React.ReactNode; screenId?: GameScreen }) {
   330→   330→  const { enableTips, currentScreen } = useGameStore();
   331→   331→  const [show, setShow] = useState(false);
   332→   332→  const [coords, setCoords] = useState({ top: 0, left: 0, arrowSide: 'top' as 'top' | 'bottom' });
   333→   333→  const tip = GAME_TIPS[tipId];
   334→   334→  const triggerRef = useRef<HTMLDivElement>(null);
   335→   335→
   336→   336→  // Check if tip should render at all
   337→   337→  const isActive = enableTips && tip && (!tip.screen || tip.screen === screenId || tip.screen === currentScreen);
   338→   338→
   339→   339→  const updatePosition = useCallback(() => {
   340→   340→    const el = triggerRef.current;
   341→   341→    if (!el) return;
   342→   342→    const rect = el.getBoundingClientRect();
   343→   343→    const tipWidth = 320; // w-80 = 20rem = 320px
   344→   344→    const tipHeight = 180; // approximate
   345→   345→    const gap = 8;
   346→   346→    const viewportHeight = window.innerHeight;
   347→   347→    const viewportWidth = window.innerWidth;
   348→   348→
   349→   349→    // Prefer showing above, but flip below if no room
   350→   350→    const showAbove = rect.top > tipHeight + gap + 50; // 50px buffer for header
   351→   351→    const arrowSide = showAbove ? 'top' : 'bottom';
   352→   352→
   353→   353→    let top: number;
   354→   354→    if (showAbove) {
   355→   355→      top = rect.top - tipHeight - gap;
   356→   356→    } else {
   357→   357→      top = rect.bottom + gap;
   358→   358→    }
   359→   359→
   360→   360→    // Center horizontally, but clamp to viewport edges
   361→   361→    let left = rect.left + rect.width / 2 - tipWidth / 2;
   362→   362→    left = Math.max(8, Math.min(left, viewportWidth - tipWidth - 8));
   363→   363→
   364→   364→    setCoords({ top: Math.max(4, top), left, arrowSide });
   365→   365→  }, []);
   366→   366→
   367→   367→  const handleEnter = useCallback(() => {
   368→   368→    if (!isActive) return;
   369→   369→    updatePosition();
   370→   370→    setShow(true);
   371→   371→  }, [isActive, updatePosition]);
   372→   372→
   373→   373→  const handleLeave = useCallback(() => setShow(false), []);
   374→   374→
   375→   375→  // Update position on scroll / resize while visible
   376→   376→  useEffect(() => {
   377→   377→    if (!show) return;
   378→   378→    updatePosition();
   379→   379→    const onScroll = () => updatePosition();
   380→   380→    const onResize = () => updatePosition();
   381→   381→    window.addEventListener('scroll', onScroll, true);
   382→   382→    window.addEventListener('resize', onResize);
   383→   383→    return () => {
   384→   384→      window.removeEventListener('scroll', onScroll, true);
   385→   385→      window.removeEventListener('resize', onResize);
   386→   386→    };
   387→   387→  }, [show, updatePosition]);
   388→   388→
   389→   389→  return (
   390→   390→    <div
   391→   391→      ref={triggerRef}
   392→   392→      className="contents"
   393→   393→      onMouseEnter={handleEnter}
   394→   394→      onMouseLeave={handleLeave}
   395→   395→    >
   396→   396→      {children}
   397→   397→      <AnimatePresence>
   398→   398→        {show && isActive && (
   399→   399→          <motion.div
   400→   400→            initial={{ opacity: 0, scale: 0.92 }}
   401→   401→            animate={{ opacity: 1, scale: 1 }}
   402→   402→            exit={{ opacity: 0, scale: 0.92 }}
   403→   403→            transition={{ duration: 0.12, ease: 'easeOut' }}
   404→   404→            style={{ position: 'fixed', top: coords.top, left: coords.left, width: 320, zIndex: 9999 }}
   405→   405→            className="pointer-events-none"
   406→   406→          >
   407→   407→            <div className="bg-popover border border-border rounded-lg shadow-xl p-3.5 text-left">
   408→   408→              {/* Header */}
   409→   409→              <div className="flex items-center gap-2 mb-2">
   410→   410→                <div className="flex items-center justify-center w-6 h-6 rounded-md bg-amber-500/15 text-amber-500 shrink-0">
   411→   411→                  <Lightbulb className="h-3.5 w-3.5" />
   412→   412→                </div>
   413→   413→                <h4 className="text-xs font-bold text-foreground">{tip.title}</h4>
   414→   414→                <Badge variant="secondary" className="text-[0.5rem] px-1.5 py-0 ml-auto bg-amber-500/10 text-amber-600 border-amber-500/20">TIP</Badge>
   415→   415→              </div>
   416→   416→              {/* Description */}
   417→   417→              <p className="text-[0.6875rem] text-muted-foreground leading-relaxed mb-2">{tip.description}</p>
   418→   418→              {/* Strategy */}
   419→   419→              <div className="border-t border-border/50 pt-2">
   420→   420→                <div className="flex items-start gap-1.5">
   421→   421→                  <Info className="h-3 w-3 text-amber-500 mt-0.5 shrink-0" />
   422→   422→                  <p className="text-[0.625rem] text-amber-600 font-medium leading-relaxed">{tip.strategy}</p>
   423→   423→                </div>
   424→   424→              </div>
   425→   425→            </div>
   426→   426→            {/* Arrow */}
   427→   427→            {coords.arrowSide === 'bottom' ? (
   428→   428→              <div className="absolute bottom-full left-[var(--arrow-x)] -translate-x-1/2 -mb-px">
   429→   429→                <div className="w-2 h-2 bg-popover border-l border-t border-border rotate-45" />
   430→   430→              </div>
   431→   431→            ) : (
   432→   432→              <div className="absolute top-full left-[var(--arrow-x)] -translate-x-1/2 -mt-px">
   433→   433→                <div className="w-2 h-2 bg-popover border-r border-b border-border rotate-45" />
   434→   434→              </div>
   435→   435→            )}
   436→   436→          </motion.div>
   437→   437→        )}
   438→   438→      </AnimatePresence>
