   523→// BUDGET SCREEN
   524→// ═══════════════════════════════════════════════════════
   525→
   526→function BudgetScreen() {
   527→  const { gameState, updateBudget, allocateBudget } = useGameStore();
   528→  const [localBudget, setLocalBudget] = useState<Record<string, number>>(() => {
   529→    if (!gameState) return {};
   530→    return Object.fromEntries(gameState.budget.items.map(i => [i.category, i.allocated]));
   531→  });
   532→
   533→  // Sync when game state changes (but not via setState in effect body)
   534→  const lastTurn = gameState?.player.turn ?? 0;
   535→  const [prevTurn, setPrevTurn] = useState(lastTurn);
   536→  if (lastTurn !== prevTurn && gameState) {
   537→    setPrevTurn(lastTurn);
   538→    const map = Object.fromEntries(gameState.budget.items.map(i => [i.category, i.allocated]));
   539→    setLocalBudget(map);
   540→  }
   541→
   542→  if (!gameState) return null;
   543→  const { budget, economic } = gameState;
   544→  const totalAllocated = Object.values(localBudget).reduce((s, v) => s + v, 0);
   545→  const deficit = totalAllocated - budget.totalRevenue;
   546→  const revenuePerCategory = budget.totalRevenue / budget.items.length;
   547→
   548→  const handleAllocate = () => {
   549→    Object.entries(localBudget).forEach(([cat, amt]) => updateBudget(cat, amt));
   550→    allocateBudget();
   551→  };
   552→
   553→  return (
   554→    <div className="space-y-6">
   555→      {/* Budget Summary */}
   556→      <HoverTip tipId="budget_overview" screenId="budget"><div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   557→        <StatCard label="Revenue" value={`$${budget.totalRevenue.toLocaleString()}M`} icon={<DollarSign className="h-4 w-4" />} color="text-green-500" />
   558→        <StatCard label="Allocated" value={`$${totalAllocated.toLocaleString()}M`} icon={<DollarSign className="h-4 w-4" />} color={deficit > 0 ? 'text-red-500' : 'text-green-500'} />
   559→        <StatCard label="Deficit" value={`$${deficit.toLocaleString()}M`} icon={<AlertTriangle className="h-4 w-4" />} color={deficit > 0 ? 'text-red-500' : 'text-green-500'} />
   560→        <StatCard label="GDP" value={`$${economic.gdp.toFixed(1)}B`} icon={<TrendingUp className="h-4 w-4" />} color={economic.gdpGrowth > 0 ? 'text-green-500' : 'text-red-500'} />
   561→      </div></HoverTip>
   562→
   563→      <div className="flex justify-end">
   564→        <HoverTip tipId="allocate_button" screenId="budget"><Button onClick={handleAllocate} size="sm" className="bg-amber-600 hover:bg-amber-700">
   565→          <Check className="h-4 w-4 mr-1" /> Allocate Budget
   566→        </Button></HoverTip>
   567→      </div>
   568→
   569→      {/* Budget Items */}
   570→      <HoverTip tipId="budget_sliders" screenId="budget"><div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   571→        {budget.items.map((item) => {
   572→          const pct = (localBudget[item.category] || 0) / budget.totalRevenue * 100;
   573→          const meetsMin = (localBudget[item.category] || 0) >= item.minimumRequired;
   574→          return (
   575→            <div key={item.category} className="bg-card border border-border rounded-lg p-4">
   576→              <div className="flex items-center justify-between mb-2">
   577→                <div className="flex items-center gap-2">
   578→                  <span className="text-lg">{item.icon}</span>
   579→                  <div>
   580→                    <h4 className="text-sm font-bold">{item.displayName}</h4>
   581→                    <p className="text-[10px] text-muted-foreground">Min: ${item.minimumRequired}M | Rec: ${item.recommended}M</p>
   582→                  </div>
   583→                </div>
   584→                <div className="text-right">
   585→                  <span className={`text-sm font-bold ${!meetsMin ? 'text-red-500' : ''}`}>
   586→                    ${localBudget[item.category]?.toLocaleString() || 0}M
   587→                  </span>
   588→                  <p className="text-[10px] text-muted-foreground">{pct.toFixed(1)}%</p>
   589→                </div>
   590→              </div>
   591→              <div className="mb-1">
   592→                <input
   593→                  type="range"
   594→                  min={0}
   595→                  max={2000}
   596→                  step={10}
   597→                  value={localBudget[item.category] || 0}
   598→                  onChange={(e) => setLocalBudget(prev => ({ ...prev, [item.category]: Number(e.target.value) }))}
   599→                  className="w-full h-2 bg-muted rounded-full appearance-none cursor-pointer accent-amber-500"
   600→                />
   601→              </div>
   602→              <div className="flex items-center gap-3 text-[10px] text-muted-foreground">
   603→                <span>Efficiency: {item.efficiency}%</span>
   604→                <span>Corruption: {item.corruptionLeakage}%</span>
   605→                {!meetsMin && <Badge variant="destructive" className="text-[9px] px-1">Below Min</Badge>}
   606→              </div>
   607→            </div>
   608→          );
   609→        })}
   610→      </div></HoverTip>
   611→    </div>
   612→  );
   613→}
