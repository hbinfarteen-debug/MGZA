   309→// HOVER TIP COMPONENT
   310→// Uses position:fixed + getBoundingClientRect to escape overflow:auto clipping.
   311→// Always renders wrapper div to prevent layout shift when toggling.
   312→// ═══════════════════════════════════════════════════════
   313→
   314→function HoverTip({ tipId, children, screenId }: { tipId: string; children: React.ReactNode; screenId?: GameScreen }) {
   315→  const { enableTips, currentScreen } = useGameStore();
   316→  const [show, setShow] = useState(false);
   317→  const [coords, setCoords] = useState({ top: 0, left: 0, arrowSide: 'top' as 'top' | 'bottom' });
   318→  const tip = GAME_TIPS[tipId];
   319→  const triggerRef = useRef<HTMLDivElement>(null);
   320→
   321→  // Check if tip should render at all
   322→  const isActive = enableTips && tip && (!tip.screen || tip.screen === screenId || tip.screen === currentScreen);
   323→
   324→  const updatePosition = useCallback(() => {
   325→    const el = triggerRef.current;
   326→    if (!el) return;
   327→    const rect = el.getBoundingClientRect();
   328→    const tipWidth = 320; // w-80 = 20rem = 320px
   329→    const tipHeight = 180; // approximate
   330→    const gap = 8;
   331→    const viewportHeight = window.innerHeight;
   332→    const viewportWidth = window.innerWidth;
   333→
   334→    // Prefer showing above, but flip below if no room
   335→    const showAbove = rect.top > tipHeight + gap + 50; // 50px buffer for header
   336→    const arrowSide = showAbove ? 'top' : 'bottom';
   337→
   338→    let top: number;
   339→    if (showAbove) {
   340→      top = rect.top - tipHeight - gap;
   341→    } else {
   342→      top = rect.bottom + gap;
   343→    }
   344→
   345→    // Center horizontally, but clamp to viewport edges
   346→    let left = rect.left + rect.width / 2 - tipWidth / 2;
   347→    left = Math.max(8, Math.min(left, viewportWidth - tipWidth - 8));
   348→
   349→    setCoords({ top: Math.max(4, top), left, arrowSide });
   350→  }, []);
   351→
   352→  const handleEnter = useCallback(() => {
   353→    if (!isActive) return;
   354→    updatePosition();
   355→    setShow(true);
   356→  }, [isActive, updatePosition]);
   357→
   358→  const handleLeave = useCallback(() => setShow(false), []);
   359→
   360→  // Update position on scroll / resize while visible
   361→  useEffect(() => {
   362→    if (!show) return;
   363→    updatePosition();
   364→    const onScroll = () => updatePosition();
   365→    const onResize = () => updatePosition();
   366→    window.addEventListener('scroll', onScroll, true);
   367→    window.addEventListener('resize', onResize);
   368→    return () => {
   369→      window.removeEventListener('scroll', onScroll, true);
   370→      window.removeEventListener('resize', onResize);
   371→    };
   372→  }, [show, updatePosition]);
   373→
   374→  return (
   375→    <div
   376→      ref={triggerRef}
   377→      className="contents"
   378→      onMouseEnter={handleEnter}
   379→      onMouseLeave={handleLeave}
   380→    >
   381→      {children}
   382→      <AnimatePresence>
   383→        {show && isActive && (
   384→          <motion.div
   385→            initial={{ opacity: 0, scale: 0.92 }}
   386→            animate={{ opacity: 1, scale: 1 }}
   387→            exit={{ opacity: 0, scale: 0.92 }}
   388→            transition={{ duration: 0.12, ease: 'easeOut' }}
   389→            style={{ position: 'fixed', top: coords.top, left: coords.left, width: 320, zIndex: 9999 }}
   390→            className="pointer-events-none"
   391→          >
   392→            <div className="bg-popover border border-border rounded-lg shadow-xl p-3.5 text-left">
   393→              {/* Header */}
   394→              <div className="flex items-center gap-2 mb-2">
   395→                <div className="flex items-center justify-center w-6 h-6 rounded-md bg-amber-500/15 text-amber-500 shrink-0">
   396→                  <Lightbulb className="h-3.5 w-3.5" />
   397→                </div>
   398→                <h4 className="text-xs font-bold text-foreground">{tip.title}</h4>
   399→                <Badge variant="secondary" className="text-[8px] px-1.5 py-0 ml-auto bg-amber-500/10 text-amber-600 border-amber-500/20">TIP</Badge>
   400→              </div>
   401→              {/* Description */}
   402→              <p className="text-[11px] text-muted-foreground leading-relaxed mb-2">{tip.description}</p>
   403→              {/* Strategy */}
   404→              <div className="border-t border-border/50 pt-2">
   405→                <div className="flex items-start gap-1.5">
   406→                  <Info className="h-3 w-3 text-amber-500 mt-0.5 shrink-0" />
   407→                  <p className="text-[10px] text-amber-600 font-medium leading-relaxed">{tip.strategy}</p>
   408→                </div>
   409→              </div>
   410→            </div>
   411→            {/* Arrow */}
   412→            {coords.arrowSide === 'bottom' ? (
   413→              <div className="absolute bottom-full left-[var(--arrow-x)] -translate-x-1/2 -mb-px">
   414→                <div className="w-2 h-2 bg-popover border-l border-t border-border rotate-45" />
   415→              </div>
   416→            ) : (
   417→              <div className="absolute top-full left-[var(--arrow-x)] -translate-x-1/2 -mt-px">
   418→                <div className="w-2 h-2 bg-popover border-r border-b border-border rotate-45" />
   419→              </div>
   420→            )}
   421→          </motion.div>
   422→        )}
   423→      </AnimatePresence>
   424→    </div>
   425→  );
   426→}

   584→// ═══════════════════════════════════════════════════════
   585→
   586→function BudgetScreen() {
   587→  const { gameState, updateBudget, allocateBudget } = useGameStore();
   588→  const [localBudget, setLocalBudget] = useState<Record<string, number>>(() => {
   589→    if (!gameState) return {};
   590→    return Object.fromEntries(gameState.budget.items.map(i => [i.category, i.allocated]));
   591→  });
   592→
   593→  // Sync when game state changes (but not via setState in effect body)
   594→  const lastTurn = gameState?.player.turn ?? 0;
   595→  const [prevTurn, setPrevTurn] = useState(lastTurn);
   596→  if (lastTurn !== prevTurn && gameState) {
   597→    setPrevTurn(lastTurn);
   598→    const map = Object.fromEntries(gameState.budget.items.map(i => [i.category, i.allocated]));
   599→    setLocalBudget(map);
   600→  }
   601→
   602→  if (!gameState) return null;
   603→  const { budget, economic } = gameState;
   604→  const totalAllocated = Object.values(localBudget).reduce((s, v) => s + v, 0);
   605→  const deficit = totalAllocated - budget.totalRevenue;
   606→  const revenuePerCategory = budget.totalRevenue / budget.items.length;
   607→
   608→  const handleAllocate = () => {
   609→    Object.entries(localBudget).forEach(([cat, amt]) => updateBudget(cat, amt));
   610→    allocateBudget();
   611→  };
   612→
   613→  return (
   614→    <div className="space-y-6">
   615→      {/* Budget Summary */}
   616→      <HoverTip tipId="budget_overview" screenId="budget"><div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   617→        <StatCard label="Revenue" value={`$${budget.totalRevenue.toLocaleString()}M`} icon={<DollarSign className="h-4 w-4" />} color="text-green-500" />
   618→        <StatCard label="Allocated" value={`$${totalAllocated.toLocaleString()}M`} icon={<DollarSign className="h-4 w-4" />} color={deficit > 0 ? 'text-red-500' : 'text-green-500'} />
   619→        <StatCard label="Deficit" value={`$${deficit.toLocaleString()}M`} icon={<AlertTriangle className="h-4 w-4" />} color={deficit > 0 ? 'text-red-500' : 'text-green-500'} />
   620→        <StatCard label="GDP" value={`$${economic.gdp.toFixed(1)}B`} icon={<TrendingUp className="h-4 w-4" />} color={economic.gdpGrowth > 0 ? 'text-green-500' : 'text-red-500'} />
   621→      </div></HoverTip>
   622→
   623→      <div className="flex justify-end">
   624→        <HoverTip tipId="allocate_button" screenId="budget"><Button onClick={handleAllocate} size="sm" className="bg-amber-600 hover:bg-amber-700">
   625→          <Check className="h-4 w-4 mr-1" /> Allocate Budget
   626→        </Button></HoverTip>
   627→      </div>
   628→
   629→      {/* Budget Items */}
   630→      <HoverTip tipId="budget_sliders" screenId="budget"><div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   631→        {budget.items.map((item) => {
   632→          const pct = (localBudget[item.category] || 0) / budget.totalRevenue * 100;
   633→          const meetsMin = (localBudget[item.category] || 0) >= item.minimumRequired;
   634→          return (
   635→            <div key={item.category} className="bg-card border border-border rounded-lg p-4">
   636→              <div className="flex items-center justify-between mb-2">
   637→                <div className="flex items-center gap-2">
   638→                  <span className="text-lg">{item.icon}</span>
   639→                  <div>
   640→                    <h4 className="text-sm font-bold">{item.displayName}</h4>
   641→                    <p className="text-[10px] text-muted-foreground">Min: ${item.minimumRequired}M | Rec: ${item.recommended}M</p>
   642→                  </div>
   643→                </div>
   644→                <div className="text-right">
   645→                  <span className={`text-sm font-bold ${!meetsMin ? 'text-red-500' : ''}`}>
   646→                    ${localBudget[item.category]?.toLocaleString() || 0}M
   647→                  </span>
   648→                  <p className="text-[10px] text-muted-foreground">{pct.toFixed(1)}%</p>
   649→                </div>
   650→              </div>
   651→              <div className="mb-1">
   652→                <input
   653→                  type="range"
   654→                  min={0}
   655→                  max={2000}
   656→                  step={10}
   657→                  value={localBudget[item.category] || 0}
   658→                  onChange={(e) => setLocalBudget(prev => ({ ...prev, [item.category]: Number(e.target.value) }))}
   659→                  className="w-full h-2 bg-muted rounded-full appearance-none cursor-pointer accent-amber-500"
   660→                />
   661→              </div>
   662→              <div className="flex items-center gap-3 text-[10px] text-muted-foreground">
   663→                <span>Efficiency: {item.efficiency}%</span>
   664→                <span>Corruption: {item.corruptionLeakage}%</span>
   665→                {!meetsMin && <Badge variant="destructive" className="text-[9px] px-1">Below Min</Badge>}
   666→              </div>
   667→            </div>
   668→          );
   669→        })}
   670→      </div></HoverTip>
   671→    </div>
   672→  );
   673→}
