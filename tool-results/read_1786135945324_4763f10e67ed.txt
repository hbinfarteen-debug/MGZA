   322→// HOVER TIP COMPONENT
   323→// Uses position:fixed + getBoundingClientRect to escape overflow:auto clipping.
   324→// Always renders wrapper div to prevent layout shift when toggling.
   325→// ═══════════════════════════════════════════════════════
   326→
   327→function HoverTip({ tipId, children, screenId }: { tipId: string; children: React.ReactNode; screenId?: GameScreen }) {
   328→  const { enableTips, currentScreen } = useGameStore();
   329→  const [show, setShow] = useState(false);
   330→  const [coords, setCoords] = useState({ top: 0, left: 0, arrowSide: 'top' as 'top' | 'bottom' });
   331→  const tip = GAME_TIPS[tipId];
   332→  const triggerRef = useRef<HTMLDivElement>(null);
   333→
   334→  // Check if tip should render at all
   335→  const isActive = enableTips && tip && (!tip.screen || tip.screen === screenId || tip.screen === currentScreen);
   336→
   337→  const updatePosition = useCallback(() => {
   338→    const el = triggerRef.current;
   339→    if (!el) return;
   340→    const rect = el.getBoundingClientRect();
   341→    const tipWidth = 320; // w-80 = 20rem = 320px
   342→    const tipHeight = 180; // approximate
   343→    const gap = 8;
   344→    const viewportHeight = window.innerHeight;
   345→    const viewportWidth = window.innerWidth;
   346→
   347→    // Prefer showing above, but flip below if no room
   348→    const showAbove = rect.top > tipHeight + gap + 50; // 50px buffer for header
   349→    const arrowSide = showAbove ? 'top' : 'bottom';
   350→
   351→    let top: number;
   352→    if (showAbove) {
   353→      top = rect.top - tipHeight - gap;
   354→    } else {
   355→      top = rect.bottom + gap;
   356→    }
   357→
   358→    // Center horizontally, but clamp to viewport edges
   359→    let left = rect.left + rect.width / 2 - tipWidth / 2;
   360→    left = Math.max(8, Math.min(left, viewportWidth - tipWidth - 8));
   361→
   362→    setCoords({ top: Math.max(4, top), left, arrowSide });
   363→  }, []);
   364→
   365→  const handleEnter = useCallback(() => {
   366→    if (!isActive) return;
   367→    updatePosition();
   368→    setShow(true);
   369→  }, [isActive, updatePosition]);
   370→
   371→  const handleLeave = useCallback(() => setShow(false), []);
   372→
   373→  // Update position on scroll / resize while visible
   374→  useEffect(() => {
   375→    if (!show) return;
   376→    updatePosition();
   377→    const onScroll = () => updatePosition();
   378→    const onResize = () => updatePosition();
   379→    window.addEventListener('scroll', onScroll, true);
   380→    window.addEventListener('resize', onResize);
   381→    return () => {
   382→      window.removeEventListener('scroll', onScroll, true);
   383→      window.removeEventListener('resize', onResize);
   384→    };
   385→  }, [show, updatePosition]);
   386→
   387→  return (
   388→    <div
   389→      ref={triggerRef}
   390→      className="contents"
   391→      onMouseEnter={handleEnter}
   392→      onMouseLeave={handleLeave}
   393→    >
   394→      {children}
   395→      <AnimatePresence>
   396→        {show && isActive && (
   397→          <motion.div
   398→            initial={{ opacity: 0, scale: 0.92 }}
   399→            animate={{ opacity: 1, scale: 1 }}
   400→            exit={{ opacity: 0, scale: 0.92 }}
   401→            transition={{ duration: 0.12, ease: 'easeOut' }}
   402→            style={{ position: 'fixed', top: coords.top, left: coords.left, width: 320, zIndex: 9999 }}
   403→            className="pointer-events-none"
   404→          >
   405→            <div className="bg-popover border border-border rounded-lg shadow-xl p-3.5 text-left">
   406→              {/* Header */}
   407→              <div className="flex items-center gap-2 mb-2">
   408→                <div className="flex items-center justify-center w-6 h-6 rounded-md bg-amber-500/15 text-amber-500 shrink-0">
   409→                  <Lightbulb className="h-3.5 w-3.5" />
   410→                </div>
   411→                <h4 className="text-xs font-bold text-foreground">{tip.title}</h4>
   412→                <Badge variant="secondary" className="text-[0.5rem] px-1.5 py-0 ml-auto bg-amber-500/10 text-amber-600 border-amber-500/20">TIP</Badge>
   413→              </div>
   414→              {/* Description */}
   415→              <p className="text-[0.6875rem] text-muted-foreground leading-relaxed mb-2">{tip.description}</p>
   416→              {/* Strategy */}
   417→              <div className="border-t border-border/50 pt-2">
   418→                <div className="flex items-start gap-1.5">
   419→                  <Info className="h-3 w-3 text-amber-500 mt-0.5 shrink-0" />
   420→                  <p className="text-[0.625rem] text-amber-600 font-medium leading-relaxed">{tip.strategy}</p>
   421→                </div>
   422→              </div>
   423→            </div>
   424→            {/* Arrow */}
   425→            {coords.arrowSide === 'bottom' ? (
   426→              <div className="absolute bottom-full left-[var(--arrow-x)] -translate-x-1/2 -mb-px">
   427→                <div className="w-2 h-2 bg-popover border-l border-t border-border rotate-45" />
   428→              </div>
   429→            ) : (
   430→              <div className="absolute top-full left-[var(--arrow-x)] -translate-x-1/2 -mt-px">
   431→                <div className="w-2 h-2 bg-popover border-r border-b border-border rotate-45" />
   432→              </div>
   433→            )}
   434→          </motion.div>
   435→        )}
   436→      </AnimatePresence>
   437→    </div>
   438→  );
   439→}

   596→// BUDGET SCREEN
   597→// ═══════════════════════════════════════════════════════
   598→
   599→function BudgetScreen() {
   600→  const { gameState, updateBudget, allocateBudget } = useGameStore();
   601→  const [localBudget, setLocalBudget] = useState<Record<string, number>>(() => {
   602→    if (!gameState) return {};
   603→    return Object.fromEntries(gameState.budget.items.map(i => [i.category, i.allocated]));
   604→  });
   605→
   606→  // Sync when game state changes (but not via setState in effect body)
   607→  const lastTurn = gameState?.player.turn ?? 0;
   608→  const [prevTurn, setPrevTurn] = useState(lastTurn);
   609→  if (lastTurn !== prevTurn && gameState) {
   610→    setPrevTurn(lastTurn);
   611→    const map = Object.fromEntries(gameState.budget.items.map(i => [i.category, i.allocated]));
   612→    setLocalBudget(map);
   613→  }
   614→
   615→  if (!gameState) return null;
   616→  const { budget, economic } = gameState;
   617→  const totalAllocated = Object.values(localBudget).reduce((s, v) => s + v, 0);
   618→  const deficit = totalAllocated - budget.totalRevenue;
   619→  const revenuePerCategory = budget.totalRevenue / budget.items.length;
   620→
   621→  const handleAllocate = () => {
   622→    Object.entries(localBudget).forEach(([cat, amt]) => updateBudget(cat, amt));
   623→    allocateBudget();
   624→  };
   625→
   626→  return (
   627→    <div className="space-y-6">
   628→      {/* Budget Summary */}
   629→      <HoverTip tipId="budget_overview" screenId="budget"><div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   630→        <StatCard label="Revenue" value={`$${budget.totalRevenue.toLocaleString()}M`} icon={<DollarSign className="h-4 w-4" />} color="text-green-500" />
   631→        <StatCard label="Allocated" value={`$${totalAllocated.toLocaleString()}M`} icon={<DollarSign className="h-4 w-4" />} color={deficit > 0 ? 'text-red-500' : 'text-green-500'} />
   632→        <StatCard label="Deficit" value={`$${deficit.toLocaleString()}M`} icon={<AlertTriangle className="h-4 w-4" />} color={deficit > 0 ? 'text-red-500' : 'text-green-500'} />
   633→        <StatCard label="GDP" value={`$${economic.gdp.toFixed(1)}B`} icon={<TrendingUp className="h-4 w-4" />} color={economic.gdpGrowth > 0 ? 'text-green-500' : 'text-red-500'} />
   634→      </div></HoverTip>
   635→
   636→      <div className="flex justify-end">
   637→        <HoverTip tipId="allocate_button" screenId="budget"><Button onClick={handleAllocate} size="sm" className="bg-amber-600 hover:bg-amber-700">
   638→          <Check className="h-4 w-4 mr-1" /> Allocate Budget
   639→        </Button></HoverTip>
   640→      </div>
   641→
   642→      {/* Budget Items */}
   643→      <HoverTip tipId="budget_sliders" screenId="budget"><div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   644→        {budget.items.map((item) => {
   645→          const pct = (localBudget[item.category] || 0) / budget.totalRevenue * 100;
   646→          const meetsMin = (localBudget[item.category] || 0) >= item.minimumRequired;
   647→          return (
   648→            <div key={item.category} className="bg-card border border-border rounded-lg p-4">
   649→              <div className="flex items-center justify-between mb-2">
   650→                <div className="flex items-center gap-2">
   651→                  <span className="text-lg">{item.icon}</span>
   652→                  <div>
   653→                    <h4 className="text-sm font-bold">{item.displayName}</h4>
   654→                    <p className="text-[0.625rem] text-muted-foreground">Min: ${item.minimumRequired}M | Rec: ${item.recommended}M</p>
   655→                  </div>
   656→                </div>
   657→                <div className="text-right">
   658→                  <span className={`text-sm font-bold ${!meetsMin ? 'text-red-500' : ''}`}>
   659→                    ${localBudget[item.category]?.toLocaleString() || 0}M
   660→                  </span>
   661→                  <p className="text-[0.625rem] text-muted-foreground">{pct.toFixed(1)}%</p>
   662→                </div>
   663→              </div>
   664→              <div className="mb-1">
   665→                <input
   666→                  type="range"
   667→                  min={0}
   668→                  max={2000}
   669→                  step={10}
   670→                  value={localBudget[item.category] || 0}
   671→                  onChange={(e) => setLocalBudget(prev => ({ ...prev, [item.category]: Number(e.target.value) }))}
   672→                  className="w-full h-2 bg-muted rounded-full appearance-none cursor-pointer accent-amber-500"
   673→                />
   674→              </div>
   675→              <div className="flex items-center gap-3 text-[0.625rem] text-muted-foreground">
   676→                <span>Efficiency: {item.efficiency}%</span>
   677→                <span>Corruption: {item.corruptionLeakage}%</span>
   678→                {!meetsMin && <Badge variant="destructive" className="text-[0.5625rem] px-1">Below Min</Badge>}
   679→              </div>
   680→            </div>
   681→          );
   682→        })}
   683→      </div></HoverTip>
   684→    </div>
   685→  );
   686→}
