   324→// HOVER TIP COMPONENT
   325→// Uses position:fixed + getBoundingClientRect to escape overflow:auto clipping.
   326→// Always renders wrapper div to prevent layout shift when toggling.
   327→// ═══════════════════════════════════════════════════════
   328→
   329→function HoverTip({ tipId, children, screenId }: { tipId: string; children: React.ReactNode; screenId?: GameScreen }) {
   330→  const { enableTips, currentScreen } = useGameStore();
   331→  const { getTip } = useTranslation();
   332→  const [show, setShow] = useState(false);
   333→  const [coords, setCoords] = useState({ top: 0, left: 0, arrowSide: 'top' as 'top' | 'bottom' });
   334→  const gameTip = GAME_TIPS[tipId];
   335→  const isActive = enableTips && gameTip && (!gameTip.screen || gameTip.screen === screenId || gameTip.screen === currentScreen);
   336→  const tip = getTip(tipId);
   337→  const triggerRef = useRef<HTMLDivElement>(null);
   338→
   339→  // Check if tip should render at all
   340→  // isActive already computed above using gameTip for screen check, tip for display
   341→
   342→  const updatePosition = useCallback(() => {
   343→    const el = triggerRef.current;
   344→    if (!el) return;
   345→    const rect = el.getBoundingClientRect();
   346→    const tipWidth = 320; // w-80 = 20rem = 320px
   347→    const tipHeight = 180; // approximate
   348→    const gap = 8;
   349→    const viewportHeight = window.innerHeight;
   350→    const viewportWidth = window.innerWidth;
   351→
   352→    // Prefer showing above, but flip below if no room
   353→    const showAbove = rect.top > tipHeight + gap + 50; // 50px buffer for header
   354→    const arrowSide = showAbove ? 'top' : 'bottom';
   355→
   356→    let top: number;
   357→    if (showAbove) {
   358→      top = rect.top - tipHeight - gap;
   359→    } else {
   360→      top = rect.bottom + gap;
   361→    }
   362→
   363→    // Center horizontally, but clamp to viewport edges
   364→    let left = rect.left + rect.width / 2 - tipWidth / 2;
   365→    left = Math.max(8, Math.min(left, viewportWidth - tipWidth - 8));
   366→
   367→    setCoords({ top: Math.max(4, top), left, arrowSide });
   368→  }, []);
   369→
   370→  const handleEnter = useCallback(() => {
   371→    if (!isActive) return;
   372→    updatePosition();
   373→    setShow(true);
   374→  }, [isActive, updatePosition]);
   375→
   376→  const handleLeave = useCallback(() => setShow(false), []);
   377→
   378→  // Update position on scroll / resize while visible
   379→  useEffect(() => {
   380→    if (!show) return;
   381→    updatePosition();
   382→    const onScroll = () => updatePosition();
   383→    const onResize = () => updatePosition();
   384→    window.addEventListener('scroll', onScroll, true);
   385→    window.addEventListener('resize', onResize);
   386→    return () => {
   387→      window.removeEventListener('scroll', onScroll, true);
   388→      window.removeEventListener('resize', onResize);
   389→    };
   390→  }, [show, updatePosition]);
   391→
   392→  return (
   393→    <div
   394→      ref={triggerRef}
   395→      className="contents"
   396→      onMouseEnter={handleEnter}
   397→      onMouseLeave={handleLeave}
   398→    >
   399→      {children}
   400→      <AnimatePresence>
   401→        {show && isActive && (
   402→          <motion.div
   403→            initial={{ opacity: 0, scale: 0.92 }}
   404→            animate={{ opacity: 1, scale: 1 }}
   405→            exit={{ opacity: 0, scale: 0.92 }}
   406→            transition={{ duration: 0.12, ease: 'easeOut' }}
   407→            style={{ position: 'fixed', top: coords.top, left: coords.left, width: 320, zIndex: 9999 }}
   408→            className="pointer-events-none"
   409→          >
   410→            <div className="bg-popover border border-border rounded-lg shadow-xl p-3.5 text-left">
   411→              {/* Header */}
   412→              <div className="flex items-center gap-2 mb-2">
   413→                <div className="flex items-center justify-center w-6 h-6 rounded-md bg-amber-500/15 text-amber-500 shrink-0">
   414→                  <Lightbulb className="h-3.5 w-3.5" />
   415→                </div>
   416→                <h4 className="text-xs font-bold text-foreground">{tip.title}</h4>
   417→              </div>
   418→              {/* Description */}
   419→              <p className="text-[0.6875rem] text-muted-foreground leading-relaxed mb-2">{tip.description}</p>
   420→              {/* Strategy */}
   421→              <div className="border-t border-border/50 pt-2">
   422→                <div className="flex items-start gap-1.5">
   423→                  <Info className="h-3 w-3 text-amber-500 mt-0.5 shrink-0" />
   424→                  <p className="text-[0.625rem] text-amber-600 font-medium leading-relaxed">{tip.strategy}</p>
   425→                </div>
   426→              </div>
   427→            </div>
   428→            {/* Arrow */}
   429→            {coords.arrowSide === 'bottom' ? (
   430→              <div className="absolute bottom-full left-[var(--arrow-x)] -translate-x-1/2 -mb-px">
   431→                <div className="w-2 h-2 bg-popover border-l border-t border-border rotate-45" />
   432→              </div>
   433→            ) : (
   434→              <div className="absolute top-full left-[var(--arrow-x)] -translate-x-1/2 -mt-px">
   435→                <div className="w-2 h-2 bg-popover border-r border-b border-border rotate-45" />
   436→              </div>
   437→            )}
   438→          </motion.div>
   439→        )}
   440→      </AnimatePresence>
   441→    </div>
   442→  );
   443→}

   602→// ═══════════════════════════════════════════════════════
   603→
   604→function BudgetScreen() {
   605→  const { gameState, updateBudget, allocateBudget } = useGameStore();
   606→  const [localBudget, setLocalBudget] = useState<Record<string, number>>(() => {
   607→    if (!gameState) return {};
   608→    return Object.fromEntries(gameState.budget.items.map(i => [i.category, i.allocated]));
   609→  });
   610→
   611→  // Sync when game state changes (but not via setState in effect body)
   612→  const lastTurn = gameState?.player.turn ?? 0;
   613→  const [prevTurn, setPrevTurn] = useState(lastTurn);
   614→  if (lastTurn !== prevTurn && gameState) {
   615→    setPrevTurn(lastTurn);
   616→    const map = Object.fromEntries(gameState.budget.items.map(i => [i.category, i.allocated]));
   617→    setLocalBudget(map);
   618→  }
   619→
   620→  if (!gameState) return null;
   621→  const { budget, economic } = gameState;
   622→  const totalAllocated = Object.values(localBudget).reduce((s, v) => s + v, 0);
   623→  const deficit = totalAllocated - budget.totalRevenue;
   624→  const revenuePerCategory = budget.totalRevenue / budget.items.length;
   625→
   626→  const handleAllocate = () => {
   627→    Object.entries(localBudget).forEach(([cat, amt]) => updateBudget(cat, amt));
   628→    allocateBudget();
   629→  };
   630→
   631→  return (
   632→    <div className="space-y-6">
   633→      {/* Budget Summary */}
   634→      <HoverTip tipId="budget_overview" screenId="budget"><div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   635→        <StatCard label="Revenue" value={`$${budget.totalRevenue.toLocaleString()}M`} icon={<DollarSign className="h-4 w-4" />} color="text-green-500" />
   636→        <StatCard label="Allocated" value={`$${totalAllocated.toLocaleString()}M`} icon={<DollarSign className="h-4 w-4" />} color={deficit > 0 ? 'text-red-500' : 'text-green-500'} />
   637→        <StatCard label="Deficit" value={`$${deficit.toLocaleString()}M`} icon={<AlertTriangle className="h-4 w-4" />} color={deficit > 0 ? 'text-red-500' : 'text-green-500'} />
   638→        <StatCard label="GDP" value={`$${economic.gdp.toFixed(1)}B`} icon={<TrendingUp className="h-4 w-4" />} color={economic.gdpGrowth > 0 ? 'text-green-500' : 'text-red-500'} />
   639→      </div></HoverTip>
   640→
   641→      <div className="flex justify-end">
   642→        <HoverTip tipId="allocate_button" screenId="budget"><Button onClick={handleAllocate} size="sm" className="bg-amber-600 hover:bg-amber-700">
   643→          <Check className="h-4 w-4 mr-1" /> Allocate Budget
   644→        </Button></HoverTip>
   645→      </div>
   646→
   647→      {/* Budget Items */}
   648→      <HoverTip tipId="budget_sliders" screenId="budget"><div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   649→        {budget.items.map((item) => {
   650→          const pct = (localBudget[item.category] || 0) / budget.totalRevenue * 100;
   651→          const meetsMin = (localBudget[item.category] || 0) >= item.minimumRequired;
   652→          return (
   653→            <div key={item.category} className="bg-card border border-border rounded-lg p-4">
   654→              <div className="flex items-center justify-between mb-2">
   655→                <div className="flex items-center gap-2">
   656→                  <span className="text-lg">{item.icon}</span>
   657→                  <div>
   658→                    <h4 className="text-sm font-bold">{item.displayName}</h4>
   659→                    <p className="text-[0.625rem] text-muted-foreground">Min: ${item.minimumRequired}M | Rec: ${item.recommended}M</p>
   660→                  </div>
   661→                </div>
   662→                <div className="text-right">
   663→                  <span className={`text-sm font-bold ${!meetsMin ? 'text-red-500' : ''}`}>
   664→                    ${localBudget[item.category]?.toLocaleString() || 0}M
   665→                  </span>
   666→                  <p className="text-[0.625rem] text-muted-foreground">{pct.toFixed(1)}%</p>
   667→                </div>
   668→              </div>
   669→              <div className="mb-1">
   670→                <input
   671→                  type="range"
   672→                  min={0}
   673→                  max={2000}
   674→                  step={10}
   675→                  value={localBudget[item.category] || 0}
   676→                  onChange={(e) => setLocalBudget(prev => ({ ...prev, [item.category]: Number(e.target.value) }))}
   677→                  className="w-full h-2 bg-muted rounded-full appearance-none cursor-pointer accent-amber-500"
   678→                />
   679→              </div>
   680→              <div className="flex items-center gap-3 text-[0.625rem] text-muted-foreground">
   681→                <span>Efficiency: {item.efficiency}%</span>
   682→                <span>Corruption: {item.corruptionLeakage}%</span>
   683→                {!meetsMin && <Badge variant="destructive" className="text-[0.5625rem] px-1">Below Min</Badge>}
   684→              </div>
   685→            </div>
   686→          );
   687→        })}
   688→      </div></HoverTip>
   689→    </div>
   690→  );
   691→}
