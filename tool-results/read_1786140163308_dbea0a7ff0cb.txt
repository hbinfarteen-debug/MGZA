   324→// HOVER TIP COMPONENT
   325→// Uses position:fixed + getBoundingClientRect to escape overflow:auto clipping.
   326→// Always renders wrapper div to prevent layout shift when toggling.
   327→// ═══════════════════════════════════════════════════════
   328→
   329→function HoverTip({ tipId, children, screenId }: { tipId: string; children: React.ReactNode; screenId?: GameScreen }) {
   330→  const { enableTips, currentScreen } = useGameStore();
   331→  const [show, setShow] = useState(false);
   332→  const [coords, setCoords] = useState({ top: 0, left: 0, arrowSide: 'top' as 'top' | 'bottom' });
   333→  const tip = GAME_TIPS[tipId];
   334→  const triggerRef = useRef<HTMLDivElement>(null);
   335→
   336→  // Check if tip should render at all
   337→  const isActive = enableTips && tip && (!tip.screen || tip.screen === screenId || tip.screen === currentScreen);
   338→
   339→  const updatePosition = useCallback(() => {
   340→    const el = triggerRef.current;
   341→    if (!el) return;
   342→    const rect = el.getBoundingClientRect();
   343→    const tipWidth = 320; // w-80 = 20rem = 320px
   344→    const tipHeight = 180; // approximate
   345→    const gap = 8;
   346→    const viewportHeight = window.innerHeight;
   347→    const viewportWidth = window.innerWidth;
   348→
   349→    // Prefer showing above, but flip below if no room
   350→    const showAbove = rect.top > tipHeight + gap + 50; // 50px buffer for header
   351→    const arrowSide = showAbove ? 'top' : 'bottom';
   352→
   353→    let top: number;
   354→    if (showAbove) {
   355→      top = rect.top - tipHeight - gap;
   356→    } else {
   357→      top = rect.bottom + gap;
   358→    }
   359→
   360→    // Center horizontally, but clamp to viewport edges
   361→    let left = rect.left + rect.width / 2 - tipWidth / 2;
   362→    left = Math.max(8, Math.min(left, viewportWidth - tipWidth - 8));
   363→
   364→    setCoords({ top: Math.max(4, top), left, arrowSide });
   365→  }, []);
   366→
   367→  const handleEnter = useCallback(() => {
   368→    if (!isActive) return;
   369→    updatePosition();
   370→    setShow(true);
   371→  }, [isActive, updatePosition]);
   372→
   373→  const handleLeave = useCallback(() => setShow(false), []);
   374→
   375→  // Update position on scroll / resize while visible
   376→  useEffect(() => {
   377→    if (!show) return;
   378→    updatePosition();
   379→    const onScroll = () => updatePosition();
   380→    const onResize = () => updatePosition();
   381→    window.addEventListener('scroll', onScroll, true);
   382→    window.addEventListener('resize', onResize);
   383→    return () => {
   384→      window.removeEventListener('scroll', onScroll, true);
   385→      window.removeEventListener('resize', onResize);
   386→    };
   387→  }, [show, updatePosition]);
   388→
   389→  return (
   390→    <div
   391→      ref={triggerRef}
   392→      className="contents"
   393→      onMouseEnter={handleEnter}
   394→      onMouseLeave={handleLeave}
   395→    >
   396→      {children}
   397→      <AnimatePresence>
   398→        {show && isActive && (
   399→          <motion.div
   400→            initial={{ opacity: 0, scale: 0.92 }}
   401→            animate={{ opacity: 1, scale: 1 }}
   402→            exit={{ opacity: 0, scale: 0.92 }}
   403→            transition={{ duration: 0.12, ease: 'easeOut' }}
   404→            style={{ position: 'fixed', top: coords.top, left: coords.left, width: 320, zIndex: 9999 }}
   405→            className="pointer-events-none"
   406→          >
   407→            <div className="bg-popover border border-border rounded-lg shadow-xl p-3.5 text-left">
   408→              {/* Header */}
   409→              <div className="flex items-center gap-2 mb-2">
   410→                <div className="flex items-center justify-center w-6 h-6 rounded-md bg-amber-500/15 text-amber-500 shrink-0">
   411→                  <Lightbulb className="h-3.5 w-3.5" />
   412→                </div>
   413→                <h4 className="text-xs font-bold text-foreground">{tip.title}</h4>
   414→                <Badge variant="secondary" className="text-[0.5rem] px-1.5 py-0 ml-auto bg-amber-500/10 text-amber-600 border-amber-500/20">TIP</Badge>
   415→              </div>
   416→              {/* Description */}
   417→              <p className="text-[0.6875rem] text-muted-foreground leading-relaxed mb-2">{tip.description}</p>
   418→              {/* Strategy */}
   419→              <div className="border-t border-border/50 pt-2">
   420→                <div className="flex items-start gap-1.5">
   421→                  <Info className="h-3 w-3 text-amber-500 mt-0.5 shrink-0" />
   422→                  <p className="text-[0.625rem] text-amber-600 font-medium leading-relaxed">{tip.strategy}</p>
   423→                </div>
   424→              </div>
   425→            </div>
   426→            {/* Arrow */}
   427→            {coords.arrowSide === 'bottom' ? (
   428→              <div className="absolute bottom-full left-[var(--arrow-x)] -translate-x-1/2 -mb-px">
   429→                <div className="w-2 h-2 bg-popover border-l border-t border-border rotate-45" />
   430→              </div>
   431→            ) : (
   432→              <div className="absolute top-full left-[var(--arrow-x)] -translate-x-1/2 -mt-px">
   433→                <div className="w-2 h-2 bg-popover border-r border-b border-border rotate-45" />
   434→              </div>
   435→            )}
   436→          </motion.div>
   437→        )}
   438→      </AnimatePresence>
   439→    </div>
   440→  );
   441→}

   598→// BUDGET SCREEN
   599→// ═══════════════════════════════════════════════════════
   600→
   601→function BudgetScreen() {
   602→  const { gameState, updateBudget, allocateBudget } = useGameStore();
   603→  const [localBudget, setLocalBudget] = useState<Record<string, number>>(() => {
   604→    if (!gameState) return {};
   605→    return Object.fromEntries(gameState.budget.items.map(i => [i.category, i.allocated]));
   606→  });
   607→
   608→  // Sync when game state changes (but not via setState in effect body)
   609→  const lastTurn = gameState?.player.turn ?? 0;
   610→  const [prevTurn, setPrevTurn] = useState(lastTurn);
   611→  if (lastTurn !== prevTurn && gameState) {
   612→    setPrevTurn(lastTurn);
   613→    const map = Object.fromEntries(gameState.budget.items.map(i => [i.category, i.allocated]));
   614→    setLocalBudget(map);
   615→  }
   616→
   617→  if (!gameState) return null;
   618→  const { budget, economic } = gameState;
   619→  const totalAllocated = Object.values(localBudget).reduce((s, v) => s + v, 0);
   620→  const deficit = totalAllocated - budget.totalRevenue;
   621→  const revenuePerCategory = budget.totalRevenue / budget.items.length;
   622→
   623→  const handleAllocate = () => {
   624→    Object.entries(localBudget).forEach(([cat, amt]) => updateBudget(cat, amt));
   625→    allocateBudget();
   626→  };
   627→
   628→  return (
   629→    <div className="space-y-6">
   630→      {/* Budget Summary */}
   631→      <HoverTip tipId="budget_overview" screenId="budget"><div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   632→        <StatCard label="Revenue" value={`$${budget.totalRevenue.toLocaleString()}M`} icon={<DollarSign className="h-4 w-4" />} color="text-green-500" />
   633→        <StatCard label="Allocated" value={`$${totalAllocated.toLocaleString()}M`} icon={<DollarSign className="h-4 w-4" />} color={deficit > 0 ? 'text-red-500' : 'text-green-500'} />
   634→        <StatCard label="Deficit" value={`$${deficit.toLocaleString()}M`} icon={<AlertTriangle className="h-4 w-4" />} color={deficit > 0 ? 'text-red-500' : 'text-green-500'} />
   635→        <StatCard label="GDP" value={`$${economic.gdp.toFixed(1)}B`} icon={<TrendingUp className="h-4 w-4" />} color={economic.gdpGrowth > 0 ? 'text-green-500' : 'text-red-500'} />
   636→      </div></HoverTip>
   637→
   638→      <div className="flex justify-end">
   639→        <HoverTip tipId="allocate_button" screenId="budget"><Button onClick={handleAllocate} size="sm" className="bg-amber-600 hover:bg-amber-700">
   640→          <Check className="h-4 w-4 mr-1" /> Allocate Budget
   641→        </Button></HoverTip>
   642→      </div>
   643→
   644→      {/* Budget Items */}
   645→      <HoverTip tipId="budget_sliders" screenId="budget"><div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   646→        {budget.items.map((item) => {
   647→          const pct = (localBudget[item.category] || 0) / budget.totalRevenue * 100;
   648→          const meetsMin = (localBudget[item.category] || 0) >= item.minimumRequired;
   649→          return (
   650→            <div key={item.category} className="bg-card border border-border rounded-lg p-4">
   651→              <div className="flex items-center justify-between mb-2">
   652→                <div className="flex items-center gap-2">
   653→                  <span className="text-lg">{item.icon}</span>
   654→                  <div>
   655→                    <h4 className="text-sm font-bold">{item.displayName}</h4>
   656→                    <p className="text-[0.625rem] text-muted-foreground">Min: ${item.minimumRequired}M | Rec: ${item.recommended}M</p>
   657→                  </div>
   658→                </div>
   659→                <div className="text-right">
   660→                  <span className={`text-sm font-bold ${!meetsMin ? 'text-red-500' : ''}`}>
   661→                    ${localBudget[item.category]?.toLocaleString() || 0}M
   662→                  </span>
   663→                  <p className="text-[0.625rem] text-muted-foreground">{pct.toFixed(1)}%</p>
   664→                </div>
   665→              </div>
   666→              <div className="mb-1">
   667→                <input
   668→                  type="range"
   669→                  min={0}
   670→                  max={2000}
   671→                  step={10}
   672→                  value={localBudget[item.category] || 0}
   673→                  onChange={(e) => setLocalBudget(prev => ({ ...prev, [item.category]: Number(e.target.value) }))}
   674→                  className="w-full h-2 bg-muted rounded-full appearance-none cursor-pointer accent-amber-500"
   675→                />
   676→              </div>
   677→              <div className="flex items-center gap-3 text-[0.625rem] text-muted-foreground">
   678→                <span>Efficiency: {item.efficiency}%</span>
   679→                <span>Corruption: {item.corruptionLeakage}%</span>
   680→                {!meetsMin && <Badge variant="destructive" className="text-[0.5625rem] px-1">Below Min</Badge>}
   681→              </div>
   682→            </div>
   683→          );
   684→        })}
   685→      </div></HoverTip>
   686→    </div>
   687→  );
   688→}

  1255→// ═══════════════════════════════════════════════════════
  1256→
  1257→function ElectionsScreen() {
  1258→  const { gameState } = useGameStore();
  1259→
  1260→  const election = gameState ? gameState.elections[gameState.elections.length - 1] : null;
  1261→
  1262→  // Confetti on election win — must be before any early returns (hooks rule)
  1263→  useEffect(() => {
  1264→    if (!election?.isOver || !election.playerWon) return;
  1265→    // Dynamic import for client-side only library
  1266→    import('canvas-confetti').then((mod) => {
  1267→      const confetti = mod.default;
  1268→      const fire = () => {
  1269→        confetti({
  1270→          particleCount: 60,
  1271→          spread: 70,
  1272→          origin: { y: 0.6 },
  1273→          colors: ['#2E8B37', '#E8A817', '#CC2936', '#FAFAF7'],
  1274→        });
  1275→      };
  1276→      fire();
  1277→      setTimeout(fire, 300);
  1278→      setTimeout(fire, 700);
  1279→    });
  1280→  }, [election?.isOver, election?.playerWon]);
  1281→
  1282→  if (!gameState || !election) return null;
  1283→

  1964→// ── useTilt hook — 3D mouse-follow tilt effect ───────────
  1965→function useTilt() {
  1966→  const ref = useRef<HTMLDivElement>(null);
  1967→
  1968→  const handleMouseMove = useCallback((e: React.MouseEvent<HTMLDivElement>) => {
  1969→    const el = ref.current;
  1970→    if (!el) return;
  1971→    const rect = el.getBoundingClientRect();
  1972→    const x = (e.clientX - rect.left) / rect.width;
  1973→    const y = (e.clientY - rect.top) / rect.height;
  1974→    const rotateX = (0.5 - y) * 12;
  1975→    const rotateY = (x - 0.5) * 12;
  1976→    el.style.transform = `perspective(800px) rotateX(${rotateX}deg) rotateY(${rotateY}deg) scale(1.02)`;
  1977→  }, []);
  1978→
  1979→  const handleMouseLeave = useCallback(() => {
  1980→    const el = ref.current;
  1981→    if (!el) return;
  1982→    el.style.transform = 'perspective(800px) rotateX(0deg) rotateY(0deg) scale(1)';
  1983→  }, []);
  1984→
  1985→  return { ref, handleMouseMove, handleMouseLeave };
  1986→}
