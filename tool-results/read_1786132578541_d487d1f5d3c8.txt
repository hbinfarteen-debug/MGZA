    17→function clamp(val: number, min: number, max: number): number {
    18→  return Math.max(min, Math.min(max, val));
    19→}
    20→
    21→function randomRange(min: number, max: number): number {
    22→  return Math.random() * (max - min) + min;
    23→}
    24→
    25→function randomInt(min: number, max: number): number {
    26→  return Math.floor(randomRange(min, max + 1));
    27→}

    29→function weightedRandom(items: { item: string; weight: number }[]): string {
    30→  const totalWeight = items.reduce((sum, i) => sum + i.weight, 0);
    31→  let r = Math.random() * totalWeight;
    32→  for (const i of items) {
    33→    r -= i.weight;
    34→    if (r <= 0) return i.item;
    35→  }
    36→  return items[items.length - 1].item;
    37→}

   230→
   231→function generateNews(state: GameState): NewsArticle[] {
   232→  const news: NewsArticle[] = [];
   233→  const month = MONTH_NAMES[state.player.month - 1];
   234→  const year = state.player.year;
   235→
   236→  // Economic news
   237→  if (state.economic.inflation > 40) {
   238→    news.push({
   239→      id: uid(), headline: `Inflation Surges to ${state.economic.inflation.toFixed(1)}%`,
   240→      subheadline: 'Prices of basic commodities continue to climb as purchasing power erodes',
   241→      body: `The cost of living has become increasingly difficult for ordinary citizens. The central bank reports inflation at ${state.economic.inflation.toFixed(1)}%, driven by currency depreciation and supply chain disruptions.`,
   242→      category: 'economic', turn: state.player.turn, month: state.player.month, year,
   243→      sentiment: 'negative', impact: -3, isBreaking: state.economic.inflation > 60,
   244→    });
   245→  }
   246→
   247→  if (state.economic.gdpGrowth > 5) {
   248→    news.push({
   249→      id: uid(), headline: `Economy Grows ${state.economic.gdpGrowth.toFixed(1)}% — Fastest in Region`,
   250→      subheadline: 'Strong GDP growth attributed to mining exports and agricultural recovery',
   251→      body: `Economic growth has exceeded expectations, reaching ${state.economic.gdpGrowth.toFixed(1)}%. Analysts credit improved mining output and a favorable agricultural season.`,
   252→      category: 'economic', turn: state.player.turn, month: state.player.month, year,
   253→      sentiment: 'positive', impact: 5, isBreaking: false,
   254→    });
   255→  }
   256→
   257→  // Infrastructure news
   258→  const completedProjects = state.projects.filter(p => p.status === 'completed' && p.completedTurn === state.player.turn);
   259→  for (const proj of completedProjects) {
   260→    news.push({
   261→      id: uid(), headline: `${proj.name} Completed`,
   262→      subheadline: `New ${proj.category} infrastructure now operational in ${proj.province}`,
   263→      body: `The government has completed construction of ${proj.name} at a cost of $${proj.cost}M. The project is expected to create ${proj.employmentCreated} jobs and improve services for local residents.`,
   264→      category: 'infrastructure', turn: state.player.turn, month: state.player.month, year,
   265→      sentiment: 'positive', impact: 3, isBreaking: false,
   266→    });
   267→  }
   268→

   332→export function simulateTurn(state: GameState): GameState {
   333→  const newState = JSON.parse(JSON.stringify(state)) as GameState;
   334→  const { player, economic, energy, water, infrastructure, national, citizenSatisfaction, publicServices, corruption, parliament, trade, commodities } = newState;
   335→  const season = SEASON_FROM_MONTH[player.month];
   336→
   337→  // ─── Advance Time ───
   338→  player.turn += 1;
   339→  player.month += 1;
   340→  if (player.month > 12) {
   341→    player.month = 1;
   342→    player.year += 1;
   343→    // Fiscal year budget processing
   344→    processBudgetFiscalYear(newState);
   345→  }
   346→
   347→  // ─── Economy Simulation ───
   348→  simulateEconomy(newState, season);
   349→
   350→  // ─── Energy Simulation ───
   351→  simulateEnergy(newState, season);
   352→
   353→  // ─── Water Simulation ───
   354→  simulateWater(newState, season);
   355→
   356→  // ─── Infrastructure Decay ───
   357→  simulateInfrastructure(newState);
   358→
   359→  // ─── Population ───
   360→  simulatePopulation(newState);
   361→
   362→  // ─── Corruption ───
   363→  simulateCorruption(newState);
   364→
   365→  // ─── Public Services & Satisfaction ───
   366→  simulateServices(newState);
   367→
   368→  // ─── Province Updates ───
   369→  simulateProvinces(newState);
   370→
   371→  // ─── Parliament ───
   372→  simulateParliament(newState);
   373→
   374→  // ─── Commodity Prices ───
   375→  simulateCommodities(newState);
   376→
   377→  // ─── Projects Progress ───
   378→  simulateProjects(newState);
   379→
   380→  // ─── Faction Dynamics ───
   381→  simulateFactions(newState);
   382→
   383→  // ─── Check for Game Over ───
   384→  checkGameOver(newState);
   385→
   386→  // ─── Generate Events ───
   387→  const newEvents = generateEvents(newState);
   388→  newState.events = [...newState.events.filter(e => !e.resolved), ...newEvents];
   389→
   390→  // ─── Generate News ───
   391→  const newNews = generateNews(newState);
   392→  newState.newsHistory = [...newNews, ...newState.newsHistory].slice(0, 100);
   393→
   394→  // ─── Update Game Log ───
   395→  newState.gameLog.push(`${MONTH_NAMES[player.month - 1]} ${player.year}: Turn ${player.turn} processed.`);
   396→
   397→  // ─── Record Historical Data ───
   398→  if (!newState.decisionHistory) newState.decisionHistory = [];
   399→  newState.decisionHistory.push({
   400→    turn: player.turn,
   401→    month: player.month,
   402→    year: player.year,
   403→    category: 'auto',
   404→    decision: 'Monthly simulation',
   405→    reasoning: 'Automatic monthly processing',
   406→    effects: [],
   407→  });
   408→
   409→  return newState;
   410→}

   412→function simulateEconomy(state: GameState, season: string): void {
   413→  const { economic, budget } = state;
   414→
   415→  // GDP response to conditions
   416→  let gdpDelta = 0;
   417→  if (economic.investorConfidence > 60) gdpDelta += 0.3;
   418→  if (economic.investorConfidence < 30) gdpDelta -= 0.4;
   419→  if (state.energy.loadSheddingHoursPerDay > 8) gdpDelta -= 0.5;
   420→  if (state.energy.loadSheddingHoursPerDay < 3) gdpDelta += 0.2;
   421→  if (state.infrastructure.roadQuality > 60) gdpDelta += 0.2;
   422→  if (state.infrastructure.roadQuality < 30) gdpDelta -= 0.3;
   423→
   424→  // Budget impact on GDP
   425→  const totalSpending = budget.items.reduce((sum, item) => sum + item.allocated, 0);
   426→  if (totalSpending > budget.totalRevenue * 1.1) gdpDelta += 0.1; // deficit spending can stimulate
   427→  if (totalSpending < budget.totalRevenue * 0.8) gdpDelta -= 0.15; // austerity slows
   428→
   429→  // Agriculture seasonal bonus
   430→  if (season === 'summer' && state.water.rainfallIndex > 50) gdpDelta += 0.15;
   431→
   432→  // Commodity prices impact
   433→  if (state.commodities.gold > 2000) gdpDelta += 0.2;
   434→  if (state.commodities.platinum > 1000) gdpDelta += 0.1;
   435→
   436→  economic.gdpGrowth = clamp(economic.gdpGrowth + randomRange(-0.3, 0.3) + gdpDelta * 0.1, -15, 20);
   437→  economic.gdp = Math.max(1, economic.gdp * (1 + economic.gdpGrowth / 100 / 12));
   438→  economic.gdpPerCapita = (economic.gdp * 1e9) / state.national.population;
   439→
   440→  // Inflation
   441→  let inflationDelta = 0;
   442→  if (budget.totalAllocated > budget.totalRevenue) inflationDelta += 0.5;
   443→  if (economic.moneySupply > 100) inflationDelta += 0.3;
   444→  if (economic.exchangeRate > 5000) inflationDelta += 0.5;
   445→  if (state.energy.loadSheddingHoursPerDay > 10) inflationDelta += 0.3;
   446→
   447→  economic.inflation = clamp(economic.inflation + inflationDelta + randomRange(-1, 1), -5, 500);
   448→
   449→  // Exchange rate
   450→  let exchangeDelta = 0;
   451→  if (economic.inflation > 30) exchangeDelta += 50;
   452→  if (economic.inflation > 50) exchangeDelta += 100;
   453→  if (economic.foreignReserves > 1) exchangeDelta -= 30;
   454→  if (economic.foreignReserves < 0.3) exchangeDelta += 80;
   455→  if (economic.investorConfidence > 50) exchangeDelta -= 20;
   456→
   457→  economic.exchangeRate = Math.max(1, economic.exchangeRate + exchangeDelta + randomRange(-50, 50));
   458→
   459→  // Black market
   460→  economic.blackMarketPremium = clamp(
   461→    (economic.exchangeRate / (economic.exchangeRate * 0.7 + 100)) * 100 + randomRange(-5, 5),
   462→    5, 200
   463→  );
   464→
   465→  // Debt
   466→  if (budget.totalAllocated > budget.totalRevenue) {
   467→    economic.governmentDebt += (budget.totalAllocated - budget.totalRevenue) / 1000;
   468→  }
   469→  economic.debtToGdp = clamp((economic.governmentDebt / economic.gdp) * 100, 0, 200);
   470→
   471→  // Tax revenue
   472→  economic.taxRevenue = economic.gdp * 0.2 * (1 - economic.informalEconomySize / 200);
   473→  economic.governmentRevenue = economic.taxRevenue * 1.3;
   474→
   475→  // Unemployment
   476→  let unemploymentDelta = 0;
   477→  if (economic.gdpGrowth > 5) unemploymentDelta -= 0.3;
   478→  if (economic.gdpGrowth < 0) unemploymentDelta += 0.5;
   479→  const activeProjects = state.projects.filter(p => p.status === 'in_progress').length;
   480→  unemploymentDelta -= activeProjects * 0.05;
   481→
   482→  economic.unemploymentRate = clamp(economic.unemploymentRate + unemploymentDelta + randomRange(-0.3, 0.3), 5, 80);
   483→  economic.youthUnemployment = clamp(economic.youthUnemployment + randomRange(-0.5, 0.5), economic.unemploymentRate * 1.3, 95);
   484→
   485→  // Consumer/investor confidence
   486→  economic.consumerConfidence = clamp(economic.consumerConfidence + (economic.inflation < 15 ? 1 : -1) + (economic.unemploymentRate < 20 ? 1 : -0.5) + randomRange(-2, 2), 5, 95);
   487→  economic.investorConfidence = clamp(economic.investorConfidence + (economic.gdpGrowth > 3 ? 0.5 : -0.5) + (state.corruption.nationalLevel < 40 ? 0.5 : -0.5) + randomRange(-1, 1), 5, 95);
   488→
   489→  // Informal economy
   490→  economic.informalEconomySize = clamp(economic.informalEconomySize + (economic.unemploymentRate > 40 ? 0.5 : -0.2) + randomRange(-0.3, 0.3), 20, 85);
   491→
   492→  // Trade
   493→  state.trade.exports = economic.gdp * 0.25 * (1 + (state.commodities.gold - 1950) / 5000);
   494→  state.trade.imports = economic.gdp * 0.3;
   495→  state.trade.tradeBalance = state.trade.exports - state.trade.imports;
   496→  state.trade.smugglingRate = clamp(state.trade.smugglingRate + (economic.exchangeRate > 5000 ? 0.5 : -0.2) + randomRange(-0.3, 0.3), 5, 40);
   497→}

   499→function simulateEnergy(state: GameState, season: string): void {
   500→  const { energy } = state;
   501→
   502→  // Rainfall changes
   503→  if (season === 'summer') {
   504→    energy.rainfallIndex = clamp(energy.rainfallIndex + randomRange(5, 15), 0, 100);
   505→  } else if (season === 'winter') {
   506→    energy.rainfallIndex = clamp(energy.rainfallIndex + randomRange(-10, -3), 0, 100);
   507→  } else {
   508→    energy.rainfallIndex = clamp(energy.rainfallIndex + randomRange(-5, 5), 0, 100);
   509→  }
   510→
   511→  // Dam levels
   512→  if (energy.rainfallIndex > 50) {
   513→    energy.damLevel = clamp(energy.damLevel + randomRange(1, 4), 5, 100);
   514→  } else if (energy.rainfallIndex < 30) {
   515→    energy.damLevel = clamp(energy.damLevel + randomRange(-4, -1), 5, 100);
   516→  }
   517→
   518→  // Hydro output depends on dam level
   519→  energy.sources.hydroelectric.output = clamp(
   520→    energy.sources.hydroelectric.capacity * (energy.damLevel / 100) * (0.8 + Math.random() * 0.1),
   521→    0, energy.sources.hydroelectric.capacity
   522→  );
   523→
   524→  // Solar output depends on season
   525→  const solarMultiplier = season === 'winter' ? 0.9 : season === 'summer' ? 0.6 : 0.85;
   526→  energy.sources.solar.output = clamp(
   527→    energy.sources.solar.capacity * solarMultiplier,
   528→    0, energy.sources.solar.capacity
   529→  );
   530→
   531→  // Coal reliability decreases with maintenance backlog
   532→  energy.sources.coal.output = clamp(
   533→    energy.sources.coal.capacity * (1 - energy.maintenanceBacklog / 200) + randomRange(-20, 20),
   534→    energy.sources.coal.capacity * 0.1, energy.sources.coal.capacity
   535→  );
   536→
   537→  // Wind
   538→  energy.sources.wind.output = clamp(
   539→    energy.sources.wind.capacity * (0.4 + Math.random() * 0.3),
   540→    0, energy.sources.wind.capacity
   541→  );
   542→
   543→  // Diesel
   544→  energy.sources.diesel.output = clamp(
   545→    energy.sources.diesel.capacity * 0.6,
   546→    0, energy.sources.diesel.capacity
   547→  );
   548→
   549→  // Imported
   550→  const importCapacity = state.economic.foreignReserves > 0.3 ? 0.7 : 0.3;
   551→  energy.sources.imported.output = clamp(
   552→    energy.sources.imported.capacity * importCapacity + randomRange(-20, 20),
   553→    0, energy.sources.imported.capacity
   554→  );
   555→
   556→  // Independent producers
   557→  energy.sources.independent.output = clamp(
   558→    energy.sources.independent.capacity * (0.4 + Math.random() * 0.2),
   559→    0, energy.sources.independent.capacity
   560→  );
   561→
   562→  // Calculate totals
   563→  energy.totalSupply = Object.values(energy.sources).reduce((sum, s) => sum + s.output, 0);
   564→  energy.totalDemand = energy.peakDemand * (0.7 + Math.random() * 0.3);
   565→  energy.deficit = Math.max(0, energy.totalDemand - energy.totalSupply);
   566→
   567→  // Load shedding calculation
   568→  if (energy.totalSupply < energy.totalDemand) {
   569→    const shortfall = (energy.totalDemand - energy.totalSupply) / energy.totalDemand;
   570→    energy.loadSheddingStage = clamp(Math.ceil(shortfall * 10), 0, 8);
   571→    energy.loadSheddingHoursPerDay = clamp(energy.loadSheddingStage * 2.5 + shortfall * 4, 0, 20);
   572→  } else {
   573→    energy.loadSheddingStage = 0;
   574→    energy.loadSheddingHoursPerDay = clamp(energy.loadSheddingHoursPerDay - 1, 0, 20);
   575→  }
   576→
   577→  // Maintenance backlog
   578→  const energyBudget = state.budget.items.find(b => b.category === 'energy')?.allocated || 0;
   579→  energy.maintenanceBacklog = clamp(energy.maintenanceBacklog - (energyBudget > 500 ? 2 : -1) + randomRange(-1, 1), 0, 100);
   580→
   581→  // Renewable percentage
   582→  const renewable = energy.sources.solar.output + energy.sources.wind.output + energy.sources.hydroelectric.output;
   583→  energy.renewablePercentage = (renewable / energy.totalSupply) * 100;
   584→}

   586→function simulateWater(state: GameState, season: string): void {

   594→    water.droughtRisk = clamp(water.droughtRisk + randomRange(-3, 5), 0, 100);
   595→    water.floodingRisk = clamp(water.floodingRisk + randomRange(-5, 3), 0, 100);

   597→

   601→    water.reservoirLevels = clamp(water.reservoirLevels - randomRange(2, 5), 5, 100);
   602→  } else if (water.floodingRisk > 60) {
   603→    water.reservoirLevels = clamp(water.reservoirLevels + randomRange(3, 8), 5, 100);
   604→  } else {
   605→    water.reservoirLevels = clamp(water.reservoirLevels + randomRange(-2, 3), 5, 100);

   611→  water.leakageRate = clamp(100 - water.pipelineCondition + 10, 10, 70);

   622→  water.waterQuality = clamp(water.waterQuality + (waterBudget > 400 ? 0.5 : -0.5) - water.leakageRate * 0.02 + randomRange(-1, 1), 10, 95);

   853→function simulateFactions(state: GameState): void {
   854→  for (const faction of state.factions) {
   855→    const satisfaction = faction.demands.reduce((sum, demand) => {
   856→      if (demand.includes('corruption') && state.corruption.nationalLevel < 50) return sum + 1;
   857→      if (demand.includes('budget') && state.budget.totalRevenue > state.budget.totalSpent) return sum + 1;
   858→      if (demand.includes('infrastructure') && state.infrastructure.roadQuality > 50) return sum + 1;
   859→      return sum;
   860→    }, 0) / faction.demands.length * 50;
   861→
   862→    faction.supportLevel = clamp(faction.supportLevel * 0.95 + satisfaction * 0.05 + randomRange(-2, 2), 5, 80);
   863→
   864→    // Update stance
   865→    if (faction.supportLevel > 50 && faction.stance === 'opponent') faction.stance = 'neutral';
   866→    if (faction.supportLevel > 60) faction.stance = 'ally';
   867→    if (faction.supportLevel < 25) faction.stance = 'opponent';
   868→    if (faction.supportLevel < 15) faction.stance = 'rebel';
   869→  }
   870→}

   943→function checkGameOver(state: GameState): void {
   944→  const { player, economic, citizenSatisfaction } = state;
   945→
   946→  // Hyperinflation collapse
   947→  if (economic.inflation > 300) {
   948→    state.isGameOver = true;
   949→    state.gameOverReason = 'HYPERINFLATION — The economy has collapsed under uncontrolled inflation. Currency is worthless, markets have emptied, and the nation faces total economic ruin.';
   950→    return;
   951→  }
   952→
   953→  // Sovereign default
   954→  if (economic.debtToGdp > 150) {
   955→    state.isGameOver = true;
   956→    state.gameOverReason = 'SOVEREIGN DEFAULT — The nation has defaulted on its debts. International creditors have frozen all lending. Government operations are paralyzed.';
   957→    return;
   958→  }
   959→
   960→  // Popular uprising
   961→  if (citizenSatisfaction.overall < 8 && player.popularity < 10) {
   962→    state.isGameOver = true;
   963→    state.gameOverReason = 'MASS UPRISING — Citizens have taken to the streets in unprecedented numbers. The government has lost all legitimacy and control.';
   964→    return;
   965→  }
   966→
   967→  // Military coup (rare)
   968→  if (player.popularity < 15 && player.legitimacy < 15 && Math.random() < 0.02) {
   969→    state.isGameOver = true;
   970→    state.gameOverReason = 'MILITARY INTERVENTION — With the government in freefall, the military has assumed control. Your presidency is over.';
   971→    return;
   972→  }
   973→}
